// @sim-sdk 1
import { api, condition, edge, js, todo, trigger, workflow } from '@sim/workflow-sdk'

export default workflow('Fixture', {}, () => {
  const start = trigger.start('Start', { inputFormat: {} })
  const gate = condition('Gate', { if: js`true`, else: true })
  start.to(gate)
  const a = api('A', { url: 'https://a.test', method: todo() })
  edge(gate, 'source', a)
  const b = api('B', { url: 'https://b.test', method: todo() })
  gate.if.to(b)
  edge(a, 'right-legacy', b)
})
