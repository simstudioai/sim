// @sim-sdk 1
import { file, fn, trigger, workflow } from '@sim/workflow-sdk'

export default workflow('Fixture', {}, () => {
  const start = trigger.start('Start', { inputFormat: [] })
  const jsCode = fn('Js code', { language: 'javascript', code: file('./functions/js-code.js') })
  const pyCode = fn('Py code', { language: 'python', code: file('./functions/py-code.py') })
  const shCode = fn('Sh code', { language: 'shell', code: file('./functions/sh-code.sh') })

  start.to(jsCode)
  jsCode.to(pyCode)
  pyCode.to(shCode)
})
