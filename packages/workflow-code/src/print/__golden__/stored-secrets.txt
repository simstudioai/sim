// @sim-sdk 1
import { agent, env, stored, trigger, workflow } from '@sim/workflow-sdk'

export default workflow('Fixture', {}, () => {
  const start = trigger.start('Start', { inputFormat: {} })
  const envKey = agent('Env key', { model: 'gpt-5.5', apiKey: env.OPENAI_API_KEY })
  const pastedKey = agent('Pasted key', { model: 'gpt-5.5', apiKey: stored() })

  start.to(envKey, pastedKey)
})
