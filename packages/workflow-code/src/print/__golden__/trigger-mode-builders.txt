// @sim-sdk 1
import { binding, block, gmail, trigger, workflow } from '@sim/workflow-sdk'

export default workflow('Fixture', {}, () => {
  const invoiceEmail = trigger.gmail.poller('Invoice email', {
    credential: binding('finance-gmail'),
    labelIds: ['INBOX'],
    markAsRead: true,
  })
  const oddTrigger = block(
    'gmail',
    'Odd trigger',
    { selectedTriggerId: 'gmail_other', markAsRead: false },
    { triggerMode: true }
  )
  const sendSummary = gmail.sendEmail('Send summary', {
    credential: binding('finance-gmail'),
    to: 'cfo@acme.test',
    subject: 'Invoices',
    body: invoiceEmail.email.body_text,
  })

  invoiceEmail.to(sendSummary)
})
