// @sim-sdk 1
import { agent, js, loop, parallel, t, trigger, workflow } from '@sim/workflow-sdk'

export default workflow('Fixture', {}, () => {
  const start = trigger.start('Start', { inputFormat: [] })
  const outer = loop.for('Outer', { count: 3 }, (p) => {
    const fan = parallel.collection('Fan', { collection: start.rows, batchSize: 5 }, (p2) => {
      const work = agent('Work', {
        model: 'm',
        systemPrompt: t`loop ${p.index} parallel ${p2.currentItem} named ${outer.index} fan ${fan.index}`,
      })

      p2.to(work)
    })

    p.to(fan)
  })
  const sibling = loop.while(
    'Sibling',
    { condition: js`<loop.index> < 3` },
    (p) => {
      const insideSibling = agent('Inside sibling', {
        model: 'm',
        systemPrompt: 'cannot see <work.content>',
      })

      p.to(insideSibling)
    },
    { inactive: { iterations: 5, doWhileCondition: js`` } }
  )

  start.to(outer)
  outer.to(sibling)
})
