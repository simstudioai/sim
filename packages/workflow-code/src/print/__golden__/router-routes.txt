// @sim-sdk 1
import { response, router, t, trigger, workflow } from '@sim/workflow-sdk'

export default workflow('Fixture', {}, () => {
  const start = trigger.start('Start', { inputFormat: { amount: 'number', reason: 'string' } })
  const triage = router('Triage', {
    context: t`${start.amount} for ${start.reason}`,
    model: 'claude-sonnet-5-5',
    routes: { small: 'Under $100, auto-approve', review: 'Needs a manager' },
  })
  start.to(triage)
  const approve = response('Approve', {})
  triage.small.to(approve)
  const escalate = response('Escalate', {})
  triage.review.to(escalate)
})
